### Normalize a path slashes

```ts
const normalizePath = (pathname: string) => {
  // Remove leading and trailing slashes
  const normalized = pathname.replace(/^\/+|\/+$/g, "");
  // Ensure the path always starts with a slash
  return normalized ? `/${normalized}` : "/";
}

```

### Set `aria-current` on all links to the current page

```js
document.addEventListener("DOMContentLoaded", () => {
  const samePage = (a, b) => {
    const sameOrigin = a.origin === b.origin;
    const samePath = normalizePath(a.pathname) === normalizePath(b.pathname);
    const samePageParam = a.searchParams.get("page") === b.searchParams.get("page");

    return sameOrigin && samePath && samePageParam;
  };

  // Remove searchParams and hash
  const currentPage = new URL(location.href);

  document.querySelectorAll("a[href]").forEach((link) => {
    if (samePage(new URL(link.href), currentPage)) {
      link.setAttribute("aria-current", "page");
    }
  });
});
```

## CSS

### Visually hidden

```css
.visually-hidden:not(:focus, :active, :focus-within) {
  clip-path: inset(50%);
  width: 1px;
  height: 1px;
  overflow: hidden;
  position: absolute;
  white-space: nowrap;
}
```

### "Oreo" focus indicator

```css
:focus-visible,
:target:focus {
  box-shadow: 0 0 0 3px var(--background);
  outline: 2px solid var(--foreground);
  outline-offset: 2px;
}
```

### Container

Feature: prevent padding from detracting from content.

```css
.container {
  --content-max-width: 1160px;
  --padding-inline: 32px;

  width: min(var(--content-max-width), 100% - var(--padding-inline) * 2);
  margin-inline: auto;
}
```
